statements -> (statement NEWLINE+)*

statement  -> boolean_expression

boolean_expression -> arithmetic_expression ((LESSER | LESSER_EQUAL | GREATER | GREATER_EQUAL | DOUBLE_EQUAL) arithmetic_expression)*

arithmetic_expression -> term ((PLUS | HYPHEN) term)*

term       -> factor ((STAR | FWD_SLASH) factor)*
            | factor

factor     -> (PLUS | MINUS) factor
            | atom

atom       -> INTEGER
            | FLOAT
            | IDENTIFIER
            | LPAREN boolean_expression RPAREN
