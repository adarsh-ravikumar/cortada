statements -> NEWLINE* statement (NEWLINE+ statement)* NEWLINE*

statement  -> var_decl | var_assign | fn_stmt | while_stmt | if_stmt
            | return_stmt | break_stmt | continue_stmt
            | expression

var_decl   -> IDENTIFIER COLON IDENTIFIER? EQUAL expression

var_assign -> IDENTIFIER EQUAL expression

fn_stmt    -> KWRD_FN IDENTIFIER
              LPAREN (param (COMMA param)*)? COMMA? RPAREN
              (THIN_ARROW IDENTIFIER)?
              suite

param -> IDENTIFIER 
         (COLON IDENTIFIER)?
         (EQUAL expression)?

while_stmt -> KWRD_WHILE expression suite
              (KWRD_ELSE suite)?

if_stmt    -> KWRD_IF expression suite
              (KWRD_ELIF expression suite)*
              (KWRD_ELSE suite)?

suite -> COLON NEWLINE? INDENT statements DEDENT

return_stmt -> KWRD_RETURN expression?

break_stmt -> KWRD_BREAK

continue_stmt -> KWRD_CONTINUE

expression -> or_expression

or_expression -> and_expression (KWRD_OR and_expression)*

and_expression -> not_expression (KWRD_AND not_expression)*

not_expression -> (KWRD_NOT) not_expression
                | boolean_expression

boolean_expression -> arithmetic_expression ((LESSER | LESSER_EQUAL | GREATER | GREATER_EQUAL | DOUBLE_EQUAL) arithmetic_expression)*


arithmetic_expression -> term ((PLUS | HYPHEN) term)*

term       -> factor ((STAR | FWD_SLASH) factor)*
            | factor

factor     -> (PLUS | MINUS) factor
            | postfix

postfix    -> atom (call_suffix)*

call_suffix  -> LPAREN (expression (COMMA expression)*)? COMMA? RPAREN

atom       -> INTEGER
            | FLOAT
            | IDENTIFIER
            | LPAREN or_expression RPAREN
