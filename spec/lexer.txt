Tab        -> '\t'
Newline    -> '\n'
Space      -> ' '
Whitespace -> Tab | Newline | Space 

MultilineComment -> '~~~' (.|Whitespace)* '~~~'
FullLineComment  -> '~' .* Newline

Keywords   -> 'fn' | 'if' | 'elif' | 'else' | 'for' | 'while' | 'return' | 'enum' | 'struct'
Identifier -> [a-zA-Z_][a-zA-Z0-9_]*
Integer    -> ([0-9]+) ('_' [0-9]+)*
Float      -> (([0-9]+) ('_' [0-9]+)*) '.' (([0-9]+) ('_' [0-9]+)*)

Colon        -> ':'
LeftParen    -> '('
RightParen   -> ')'
LeftBracket  -> '['
RightBracket -> ']'
LeftBrace    -> '{'
RightBrace   -> '}'
DoubleStar   -> '**'
Star         -> '*'
FwdSlash     -> '/'
Plus         -> '+'
ThinArrow    -> '->'
Hyphen       -> '-'
DoubleDot    -> '..'
Dot          -> '.'
GreaterEqual -> '>='
Greater      -> '>'
LesserEqual  -> '<='
Lesser       -> '<'
DoubleEqual  -> '=='
FatArrow     -> '=>'
Equal        -> '='

DoubleQuote  -> '"'
SingleQuote  -> "'"
Backtick     -> '`'
